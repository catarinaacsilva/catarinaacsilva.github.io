\begin{frame}
\frametitle{Breve Evolução até à Cloud}
\begin{itemize}
    \setlength\itemsep{0.4cm}
    \item \textbf{Mainframes com time-sharing} (décadas de 1960--70): vários utilizadores partilham um único computador central, por turnos de tempo de processador.
    \item \textbf{Modelo cliente-servidor} (décadas de 1980--90): processamento distribuído entre servidores dedicados e computadores pessoais.
    \item \textbf{Virtualização} (anos 2000): permite correr vários sistemas operativos isolados sobre o mesmo hardware físico, preparando o terreno para a cloud.
    \item \textbf{Cloud pública}: fornecedores passam a oferecer estes recursos virtualizados como serviço, faturado por utilização, a partir de meados dos anos 2000.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{O que é a Computação na Nuvem}
\begin{itemize}
    \setlength\itemsep{0.4cm}
    \item Definição do NIST (\textit{National Institute of Standards and Technology}): modelo que permite acesso a um conjunto partilhado de recursos computacionais configuráveis.
    \item Recursos disponibilizados e libertados com esforço mínimo de gestão ou interação com o fornecedor.
    \item Assenta em cinco características essenciais, detalhadas de seguida.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{As Cinco Características Essenciais (NIST)}
\begin{itemize}
    \setlength\itemsep{0.35cm}
    \item \textbf{Self-service sob procura}: o utilizador aprovisiona recursos sem intervenção humana do lado do fornecedor.
    \item \textbf{Acesso alargado à rede}: recursos acessíveis através da rede, por mecanismos padronizados (ex.\ browser, API).
    \item \textbf{Partilha de recursos} (\textit{resource pooling}): recursos físicos partilhados entre múltiplos clientes (multi-tenancy).
    \item \textbf{Elasticidade rápida}: capacidade de aumentar ou reduzir recursos de forma quase imediata, consoante a procura.
    \item \textbf{Serviço medido}: consumo de recursos monitorizado, controlado e reportado de forma transparente (modelo \textit{pay-per-use}).
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Modelos de Serviço}
\begin{itemize}
    \setlength\itemsep{0.4cm}
    \item \textbf{IaaS} (Infrastructure as a Service): fornecimento de recursos computacionais básicos (processamento, armazenamento, rede).
    \item \textbf{PaaS} (Platform as a Service): plataforma de desenvolvimento e execução de aplicações, sem gestão direta da infraestrutura subjacente.
    \item \textbf{SaaS} (Software as a Service): aplicações completas disponibilizadas ao utilizador final através da rede.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Modelo de Responsabilidade Partilhada}
\begin{itemize}
    \setlength\itemsep{0.4cm}
    \item Em qualquer modelo de serviço, a responsabilidade pela gestão e segurança divide-se entre fornecedor e cliente.
    \item Em \textbf{IaaS}, o fornecedor garante o hardware, a virtualização e a rede; o cliente é responsável pelo sistema operativo, aplicações e dados.
    \item Em \textbf{PaaS}, o fornecedor assume também o sistema operativo e o runtime; o cliente foca-se apenas na aplicação e nos dados.
    \item Em \textbf{SaaS}, o fornecedor gere praticamente tudo; o cliente é responsável sobretudo pela configuração de acessos e pelos dados que introduz.
    \item Esta unidade curricular foca-se sobretudo na camada de infraestrutura (IaaS) e na sua administração.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Modelos de Implantação}
\begin{itemize}
    \setlength\itemsep{0.4cm}
    \item \textbf{Cloud pública:} infraestrutura partilhada, gerida por um fornecedor externo, disponível a qualquer entidade.
    \item \textbf{Cloud privada:} infraestrutura dedicada a uma única organização.
    \item \textbf{Cloud híbrida:} combinação de infraestrutura privada e pública, com interoperabilidade entre ambas.
    \item \textbf{Cloud comunitária:} infraestrutura partilhada por várias organizações com interesses comuns.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Porque Existem Datacenters}
\begin{itemize}
    \setlength\itemsep{0.4cm}
    \item Necessidade de concentrar capacidade computacional para responder a procura em escala.
    \item Consolidação de recursos: melhor aproveitamento de servidores, rede e armazenamento.
    \item Eficiência energética: ganhos de escala em arrefecimento, alimentação elétrica e gestão operacional.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Componentes Físicos de um Datacenter}
\begin{itemize}
    \setlength\itemsep{0.4cm}
    \item Racks: estruturas normalizadas para instalação de servidores e equipamento de rede.
    \item Servidores: unidades de processamento, memória e armazenamento.
    \item PDUs (Power Distribution Units): distribuição controlada de energia elétrica pelos racks.
    \item Cablagem estruturada: organização física das ligações de rede e energia.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Equipamento de Rede num Datacenter}
\begin{itemize}
    \setlength\itemsep{0.4cm}
    \item \textbf{Switches Top-of-Rack (ToR)}: ligam os servidores de um rack entre si e ao resto da rede do datacenter.
    \item \textbf{Switches/routers de agregação e core}: interligam vários racks e ligam o datacenter ao exterior.
    \item \textbf{Firewalls e load balancers de hardware}: controlam e distribuem o tráfego que entra e sai do datacenter.
    \item A organização destes equipamentos em topologias será aprofundada na Aula 02.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Requisitos de Operação \textcolor{NudeTaupe}{\small(1/2)}}
\begin{itemize}
    \setlength\itemsep{0.4cm}
    \item Energia: sistemas UPS (Uninterruptible Power Supply) e geradores de emergência.
    \item Arrefecimento: unidades CRAC/CRAH (Computer Room Air Conditioning/Handling).
    \item Organização em corredores quentes e frios para otimizar o fluxo de ar.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Requisitos de Operação \textcolor{NudeTaupe}{\small(2/2)}}
\begin{itemize}
    \setlength\itemsep{0.4cm}
    \item Rede: ligações redundantes e de elevada capacidade.
    \item Segurança física: controlo de acessos, videovigilância, deteção de incêndios.
    \item Estes requisitos condicionam a disponibilidade e a fiabilidade do datacenter.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Métricas de Eficiência}
\begin{itemize}
    \setlength\itemsep{0.4cm}
    \item \textbf{PUE} (Power Usage Effectiveness): rácio entre a energia total consumida e a energia usada pelo equipamento de TI.
    \item Um PUE próximo de 1.0 indica elevada eficiência energética.
    \item \textbf{DCiE} (Data Center infrastructure Efficiency): inverso do PUE, expresso em percentagem.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{A Infraestrutura de Laboratório desta UC}
\begin{itemize}
    \setlength\itemsep{0.4cm}
    \item Os trabalhos práticos desta UC vão correr sobre um \textbf{servidor central com Proxmox VE}, já preparado pela instituição.
    \item Proxmox VE é uma plataforma de virtualização open-source do tipo 1 (bare-metal), assente em \textbf{KVM} (máquinas virtuais) e \textbf{LXC} (contentores de sistema).
    \item Cada aluno terá acesso à interface Proxmox para criar e gerir as suas próprias VMs/contentores, tal como um administrador faria numa infraestrutura real.
    \item Este modelo permite exercitar, na prática, muitos dos conceitos de administração de recursos abordados ao longo do semestre.
\end{itemize}
\end{frame}
